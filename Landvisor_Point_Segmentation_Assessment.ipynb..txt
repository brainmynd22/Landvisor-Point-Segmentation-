        "class PartialCELoss(nn.Module):\n",
        "    \"\"\"\n",
        "    Computes Cross-Entropy loss ONLY at labeled point positions,\n",
        "    ignoring all unannotated pixels marked with ignore_index (-1).\n",
        "    \"\"\"\n",
        "    def __init__(self, ignore_index=-1):\n",
        "        super(PartialCELoss, self).__init__()\n",
        "        self.ignore_index = ignore_index\n",
        "\n",
        "    def forward(self, logits, targets):\n",
        "        # Create mask of annotated locations\n",
        "        valid_mask = (targets != self.ignore_index)\n",
        "\n",
        "        if not valid_mask.any():\n",
        "            return torch.tensor(0.0, device=logits.device, requires_grad=True)\n",
        "\n",
        "        # Reshape to [N, Num_Classes] and target to [N]\n",
        "        logits_flat = logits.permute(0, 2, 3, 1)[valid_mask]\n",
        "        targets_flat = targets[valid_mask]\n",
        "\n",
        "        return F.cross_entropy(logits_flat, targets_flat, reduction='mean')"

        "def sample_point_annotations(dense_mask, points_per_class=5, ignore_index=-1):\n",
        "    \"\"\"\n",
        "    Simulates point annotations by sampling N random pixels per present class.\n",
        "    \"\"\"\n",
        "    sparse_mask = np.full_like(dense_mask, fill_value=ignore_index)\n",
        "    unique_classes = np.unique(dense_mask)\n",
        "\n",
        "    for c in unique_classes:\n",
        "        y_indices, x_indices = np.where(dense_mask == c)\n",

        "            num_samples = min(points_per_class, len(y_indices))\n",
        "            sampled_idx = np.random.choice(len(y_indices), size=num_samples, replace=False)\n",

        "def calculate_miou(preds, targets, num_classes=2):\n",
        "    ious = []\n",
        "    preds = torch.argmax(preds, dim=1)\n",
        "    for c in range(num_classes):\n",
        "        intersection = ((preds == c) & (targets == c)).sum().item()\n",
        "        union = ((preds == c) | (targets == c)).sum().item()\n",
        "        if union == 0:\n",
        "            ious.append(1.0)\n",
        "        else:\n",
        "            ious.append(intersection / union)\n",
        "    return np.mean(ious)\n",
